              </div>
            </div>
          ))}
        </div>

        {featured && (
          <div className="mt-12 text-center">
            <Button to="/work" variant="primary" size="md" icon={ArrowUpRight}>
              View All Work
            </Button>
          </div>
        )}

        {/* Modal Inspector for Project Details */}
        {selectedProject && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/90 backdrop-blur-xl">
            <div className="relative w-full max-w-2xl max-h-[92vh] overflow-y-auto rounded-3xl bg-[#141B2B] border border-white/[0.16] p-6 sm:p-9 shadow-2xl space-y-6">
              
              {/* Modal Top Bar */}
              <div className="flex items-start justify-between">
                <div>
                  <div className="flex items-center gap-2.5 mb-2">
                    <Badge variant={selectedProject.badge === "Concept Project" ? "blue" : "violet"}>
                      {selectedProject.badge}
                    </Badge>
                    <span className="text-xs font-mono font-semibold text-[#3898FF]">
                      {selectedProject.category}
                    </span>
                  </div>
                  <h3 className="text-2xl sm:text-3xl font-extrabold text-[#F5F7FC]">
                    {selectedProject.title}
                  </h3>
                </div>
                <button
                  onClick={() => setSelectedProject(null)}
                  className="p-2.5 text-[#A8B1C2] hover:text-[#F5F7FC] rounded-xl bg-white/[0.06] hover:bg-white/[0.12] transition-colors"
                  aria-label="Close modal"
                >
                  <X className="w-5 h-5" />
                </button>
              </div>

              {/* Large Project Visual Mockup */}
              <div className="rounded-2xl overflow-hidden border border-white/[0.08]">
                <ProjectVisualMockup project={selectedProject} />
              </div>

              {/* Business-Friendly Detailed Sections */}
              <div className="space-y-4 text-xs sm:text-sm text-[#A8B1C2] leading-relaxed">
                <div>
                  <h4 className="font-mono text-xs uppercase tracking-wider text-[#F5F7FC] font-bold mb-1.5">
                    Project Overview
                  </h4>
                  <p className="p-4 rounded-xl bg-[#101522] border border-white/[0.08] text-[#F5F7FC]">
                    {selectedProject.summary}
                  </p>
                </div>

                <div>
                  <h4 className="font-mono text-xs uppercase tracking-wider text-[#A8B1C2] font-bold mb-1.5">
                    Project Objective
                  </h4>
                  <p className="p-4 rounded-xl bg-[#101522] border border-white/[0.08]">
                    {selectedProject.challenge}
                  </p>
                </div>

                <div>
                  <h4 className="font-mono text-xs uppercase tracking-wider text-[#3898FF] font-bold mb-1.5">
                    Our Approach & Strategy
                  </h4>
                  <p className="p-4 rounded-xl bg-[#3898FF]/10 border border-[#3898FF]/30 text-[#F5F7FC]">
                    {selectedProject.solution}
                  </p>
                </div>

                <div>
                  <h4 className="font-mono text-xs uppercase tracking-wider text-[#F5F7FC] font-bold mb-1.5">
                    Key Deliverables
                  </h4>
                  <ul className="space-y-2 p-4 rounded-xl bg-[#101522] border border-white/[0.08]">
                    {selectedProject.deliverables.map((item) => (
                      <li key={item} className="flex items-center gap-2.5 text-xs text-[#F5F7FC]">
                        <CheckCircle2 className="w-4 h-4 text-[#3898FF] shrink-0" />
                        <span>{item}</span>
                      </li>
                    ))}
                  </ul>
                </div>
              </div>

              {/* Clear Concept / Demonstration Disclosure Notice */}
              <div className="p-3.5 rounded-xl bg-white/[0.03] border border-white/[0.08] text-[11px] font-mono text-[#A8B1C2] leading-relaxed">
                <span className="text-[#3898FF] font-bold block mb-0.5">
                  Concept Demonstration Disclosure:
                </span>
                This showcase is an internal demonstration framework created by Navandan Emporium to demonstrate our capabilities without misrepresenting past client experience.
              </div>

              {/* Modal Bottom CTA */}
              <div className="pt-4 border-t border-white/[0.08] flex flex-col sm:flex-row items-center justify-between gap-3">
                <span className="text-xs font-mono text-[#A8B1C2]">
                  {selectedProject.badge} • Concept Preview
                </span>
                <Button
                  to="/contact"
                  variant="primary"
                  size="sm"
                  onClick={() => setSelectedProject(null)}
                >
                  Inquire Similar Solution
                </Button>
              </div>

            </div>
          </div>
        )}

      </div>
    </section>
  );
};
